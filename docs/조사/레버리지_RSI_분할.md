## 4. 용어 정의

| 말 | 뜻 |
| --- | --- |
| **RSI** | Wilder RSI(14). 첫 값은 처음 14개 등락의 단순 평균이고, 그다음은 `(이전 × 13 + 오늘) ÷ 14` 다. 종가로 계산한다 |
| **하향 돌파** | 전날 RSI 가 문턱 이상이고 오늘 RSI 가 문턱 미만인 날 |
| **재진입 — 하향 돌파만** | 첫 매수를 하향 돌파일에만 한다 |
| **재진입 — 청산 당일** | 포지션이 없고 종가 RSI 가 문턱 미만이면 산다. 장중에 판 날의 종가도 포함한다 |
| **재진입 — 다음 거래일** | 같은 조건이되 청산한 날은 보지 않고 다음 거래일 종가부터 본다 |
| **매도가 정정 — 즉시** | 추가 매수가 체결되면 곧바로 새 평단 +5% 로 매도 주문을 고친다. 그날 고가가 새 매도가 이상이면 그날 판다 |
| **매도가 정정 — 다음 날** | 추가 매수가 체결된 날에는 새 매도가로 팔리지 않는다고 본다 |
| **사건** | 앞 진입에서 30달력일 안의 재진입을 한 사건으로 묶는다. 이 30일은 측정하며 정한 값이다 |
| **보유 중 최저** | 그날 들고 있던 수량 × 장중 저가 − 원가의 최솟값(원). 첫 매수 다음 날부터 청산일까지 본다. 0 보다 크면 0 으로 적었다 |
| **익절 없이 최저** | 같은 진입에서 +5% 매도를 끄고 RSI 50 까지 버텼을 때의 보유 중 최저. 반등에 기대지 않은 크기다 |
| **기준선** | 규칙 표에서는 종가 RSI < 50 인 **모든 날**을 진입일로 같은 규칙을 돌린 성적이다(날들이 서로 겹친다). 맨몸 격자에서는 **모든 거래일**의 같은 보유 수익률이다 |
| **우연확률** | 무작위 뽑기 대조. 기준선에서 신호 수만큼 비복원으로 뽑기를 1,000번 하고(시드 0), 기준선 평균에서 신호 평균만큼 멀어진 비율을 양측으로 센다 |

## 13. 해석

**규칙의 우연확률 0.006 과 맨몸 격자의 0.183 이상은 다른 것을 묻는다.** 규칙은 「+5% 에 닿으면 판다」는 구조라, 기준선과 달라지는 것은 **신호일 뒤 변동성이 커서 +5% 에 잘 닿는 것**일 수 있다. 맨몸 격자가 평균 방향의 우위를 보이지 않으므로, 이 신호를 「반등을 맞히는 신호」로 읽을 근거는 없다. **이 해석은 추정이고, 변동성을 따로 통제해 재지 않았다.**

**승률 100% 는 「자주 조금 맞고 가끔 크게 틀리는」 모양의 앞부분일 수 있다.** 기준선 승률이 이미 91.53% 이고, 기준선의 진 거래 155건은 평균 -543,303원 · 최악 -3,250,575원이다. 신호 칸의 거래 수가 그 드문 손실을 만나기에 충분하지 않다.

**기준선과 달라 보이는 것 자체는 탈락 사유가 아니고, 같아 보이는 것도 그렇다**(루트 `CLAUDE.md`). 이 조사가 걸지 않는 이유는 §12.2 의 크기와 위험이다.

**미국 3배에서 문턱 30 의 전승을 믿지 않는 이유**는 §10.3 이다. 상관 0.944 인 두 상품의 운명이 0.2%p 차이로 갈렸다.

"""레버리지_RSI_분할 재현. 첫 인자는 시세 폴더, 없으면 storage/market."""
import sys
from pathlib import Path

import numpy as np
import pandas as pd

from verify_lab.common_constants import MARKET_DIR
from verify_lab.data.loader import load_market_csv

MARKET = Path(sys.argv[1]) if len(sys.argv) > 1 else MARKET_DIR
DATA_END = pd.Timestamp("2026-10-30")
PERIOD, CAP = 14, 10_000_000
RUNGS, W = (-0.05, -0.10), CAP / 3
TARGETS = {
    "KODEX 레버리지": "122630_max.csv",
    "KODEX 코스닥150레버리지": "233740_max.csv",
    "TQQQ": "TQQQ_adjusted_max.csv",
    "SOXL": "SOXL_adjusted_max.csv",
    "UPRO": "UPRO_adjusted_max.csv",
    "UDOW": "UDOW_adjusted_max.csv",
}


def wilder_rsi(close):
    close = np.asarray(close, dtype=float)
    delta = np.diff(close, prepend=np.nan)
    gain, loss = np.where(delta > 0, delta, 0.0), np.where(delta < 0, -delta, 0.0)
    rsi = np.full(len(close), np.nan)
    ag, al = gain[1 : PERIOD + 1].mean(), loss[1 : PERIOD + 1].mean()
    rsi[PERIOD] = 100.0 if al == 0 else 100 - 100 / (1 + ag / al)
    for i in range(PERIOD + 1, len(close)):
        ag = (ag * (PERIOD - 1) + gain[i]) / PERIOD
        al = (al * (PERIOD - 1) + loss[i]) / PERIOD
        rsi[i] = 100.0 if al == 0 else 100 - 100 / (1 + ag / al)
    return rsi


def hold(o, h, l, c, rsi, entry, first, *, fill="B", stop=None, exit50="close", take_profit=True):
    """first 일부터 판정한다. 반환: (청산 위치, 손익(원), 사유, 보유 중 최저(원), 매수 횟수)"""
    shares, cost, filled, worst, pending = W / entry, W, 1, 0.0, False
    stop_price = entry * (1 - stop) if stop else None

    def add(t):
        nonlocal shares, cost, filled
        added = False
        for k in range(filled - 1, 2):
            level = entry * (1 + RUNGS[k])
            price = o[t] if o[t] <= level else (level if l[t] <= level else None)
            if price is not None:
                shares, cost, filled, added = shares + W / price, cost + W, filled + 1, True
        return added

    for t in range(first, len(c)):
        if pending:
            return t, shares * o[t] - cost, "RSI 50 다음 날 시가", min(worst, shares * o[t] - cost), filled
        target = cost / shares * 1.05
        if take_profit and o[t] >= target:
            return t, shares * o[t] - cost, "+5% 시가", min(worst, shares * l[t] - cost), filled
        if stop_price is not None and o[t] <= stop_price:
            return t, shares * o[t] - cost, "갭손절", min(worst, shares * o[t] - cost), filled
        # 손절선에 닿는 날은 손절을 먼저 본다(보수적). 하락 경로의 추가 매수는 손절 전에 체결된다
        if stop_price is not None and l[t] <= stop_price:
            add(t)
            return t, shares * stop_price - cost, "장중손절", min(worst, shares * stop_price - cost), filled
        if take_profit and h[t] >= target:
            return t, shares * target - cost, "+5% 장중", min(worst, shares * l[t] - cost), filled
        added = add(t)
        worst = min(worst, shares * l[t] - cost)
        target = cost / shares * 1.05
        if take_profit and fill == "B" and added and h[t] >= target:
            return t, shares * target - cost, "+5% 추가 체결 당일", worst, filled
        if rsi[t] >= 50:
            if exit50 == "close":
                return t, shares * c[t] - cost, "RSI 50 종가", worst, filled
            pending = True
    return len(c) - 1, shares * c[-1] - cost, "미청산", worst, filled


def simulate(df, threshold, mode, *, entry_at="close", exit50="close", fill="B", stop=None):
    """mode: cross = 하향 돌파만 / same = 청산 당일 종가부터 / next = 청산 다음 거래일 종가부터.
    entry_at="open" 은 판정일 다음 거래일 시가에 사고, 청산한 날의 종가도 신호로 본다"""
    o, h, l, c = [df[x].to_numpy(float) for x in ("Open", "High", "Low", "Close")]
    rsi = wilder_rsi(c)
    prev = np.roll(rsi, 1)
    prev[0] = np.nan
    rows, t, last_exit = [], PERIOD + 1, -1
    while t < len(c):
        if entry_at == "open":
            signal = rsi[t] < threshold and t >= last_exit
        elif mode == "cross":
            signal = prev[t] >= threshold and rsi[t] < threshold
        else:
            signal = rsi[t] < threshold and (mode == "same" or t != last_exit)
        if not signal or t < last_exit:
            t += 1
            continue
        if entry_at == "open" and t + 1 >= len(c):
            break
        price, first, entered = (o[t + 1], t + 1, t + 1) if entry_at == "open" else (c[t], t + 1, t)
        kwargs = {"fill": fill, "stop": stop, "exit50": exit50}
        e, pnl, why, worst, filled = hold(o, h, l, c, rsi, price, first, **kwargs)
        no_tp = hold(o, h, l, c, rsi, price, first, take_profit=False, **kwargs)[3]
        rows.append(
            {
                "신호일": df.Date.iloc[t], "진입일": df.Date.iloc[entered], "진입가": price, "청산일": df.Date.iloc[e],
                "보유일": e - entered, "매수 횟수": filled, "청산 사유": why, "손익": pnl, "보유 중 최저": worst,
                "익절 없이 최저": no_tp,
            }
        )
        last_exit = e
        # 끝나지 않은 포지션은 데이터 끝까지 들고 있는 것이라 재진입이 없다
        if why == "미청산":
            break
        reenter = entry_at == "open" or (mode == "same" and not why.startswith("RSI 50"))
        t = e if reenter else e + 1
    return pd.DataFrame(rows)


def event_ids(dates):
    gaps = pd.to_datetime(pd.Series(dates)).sort_values().diff().dt.days
    return (gaps > 30).cumsum().to_numpy()


def row(trades):
    done = trades[trades["청산 사유"] != "미청산"] if len(trades) else trades
    if done.empty:
        return {"거래": 0, "사건": 0, "미청산": len(trades)}
    values = done["손익"]
    return {
        "거래": len(done), "사건": int(event_ids(done["진입일"]).max()) + 1, "승률": round((values > 0).mean() * 100, 2),
        "합계": round(values.sum()), "평균": round(values.mean()), "중앙값": round(values.median()),
        "최악": round(values.min()), "보유 중 최저": round(done["보유 중 최저"].min()),
        "익절 없이 최저": round(done["익절 없이 최저"].min()), "평균 보유": round(done["보유일"].mean(), 1),
        "최장": int(done["보유일"].max()), "세 몫": int((done["매수 횟수"] == 3).sum()),
        "첫 몫만": int((done["매수 횟수"] == 1).sum()), "RSI 청산": int(done["청산 사유"].str.startswith("RSI").sum()),
        "손절": int(done["청산 사유"].str.endswith("손절").sum()), "미청산": len(trades) - len(done),
    }


def baseline_pool(df, *, entry_at="close", exit50="close"):
    """RSI < 50 인 모든 날을 진입일로 같은 규칙을 돌린다. 끝나지 않은 진입은 뺀다"""
    o, h, l, c = [df[x].to_numpy(float) for x in ("Open", "High", "Low", "Close")]
    rsi, rows = wilder_rsi(c), []
    for s in range(PERIOD + 1, len(c)):
        if rsi[s] >= 50 or (entry_at == "open" and s + 1 >= len(c)):
            continue
        price, first = (o[s + 1], s + 1) if entry_at == "open" else (c[s], s + 1)
        e, pnl, why, worst, _ = hold(o, h, l, c, rsi, price, first, exit50=exit50)
        if why != "미청산":
            rows.append({"진입일": df.Date.iloc[s], "손익": pnl, "보유 중 최저": worst})
    return pd.DataFrame(rows)


def chance(sample, pool, seed=0, repeats=1000):
    """무작위 뽑기 대조. 기준선 평균에서 신호 평균만큼 멀어진 비율(양측 · 비복원)"""
    rng = np.random.default_rng(seed)
    draws = np.array([rng.choice(pool, len(sample), replace=False).mean() for _ in range(repeats)])
    return round(float((np.abs(draws - pool.mean()) >= abs(sample.mean() - pool.mean())).mean()), 4)

def synthetic(df, multiple):
    """1배의 전일 종가 대비 시가 · 고가 · 저가 · 종가 등락에 배수를 곱해 합성 전일 종가에 붙인다"""
    previous = df.Close.shift(1)
    close, rows = 100.0, []
    for i in range(1, len(df)):
        base = close
        row_ = {x: base * (1 + multiple * (df[x].iloc[i] / previous.iloc[i] - 1)) for x in ("Open", "High", "Low", "Close")}
        close = row_["Close"]
        rows.append({"Date": df.Date.iloc[i], **row_})
    return pd.DataFrame(rows)
